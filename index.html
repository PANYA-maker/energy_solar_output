📋 โค้ด Google Apps Script (ใส่ในส่วนขยาย > Apps Script): 
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Daily_Summary") || ss.insertSheet("Daily_Summary");
  var data = JSON.parse(e.postData.contents);
  if (data.action === "save_daily") {
    sheet.appendRow([data.date, data.pea_onpeak_kwh, data.pea_offpeak_kwh, data.pea_total_kwh, data.peak_kw, data.solar_unit, data.output_total, data.rate_per_pea, data.unit_per_output, data.cost_per_output, data.solar_ratio, new Date()]);
    return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON);
  }
}
